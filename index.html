⚙️ Google Sheets Setup

Connect to Google Sheets as a free backend. Follow all 4 steps.

STEP 1

Open sheets.google.com → create a spreadsheet named MineShift Reports.

STEP 2

Click Extensions → Apps Script. Replace code with:

const SS = SpreadsheetApp.getActiveSpreadsheet(); function doPost(e) { try { const d = JSON.parse(e.postData.contents || '{}'); const name = d.sheet || 'Reports'; let sh = SS.getSheetByName(name); if (name === 'ShiftData') { const hdr = ['Key','Date','Shift','Payload','SavedAt']; if (!sh) { sh = SS.insertSheet(name); sh.appendRow(hdr); } const row = [d.key||'', d.date||'', d.shift||'', d.payload||'', d.savedAt||new Date().toISOString()]; const rows = sh.getDataRange().getValues(); let rowNo = -1; for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(d.key)) { rowNo = i + 1; break; } } if (rowNo === -1) sh.appendRow(row); else sh.getRange(rowNo, 1, 1, row.length).setValues([row]); return out({ status: 'ok' }); } if (!sh) { sh = SS.insertSheet(name); sh.appendRow(Object.keys(d.row)); } sh.appendRow(Object.values(d.row)); return out({ status: 'ok' }); } catch (err) { return out({ status: 'error', msg: err.message }); } } function doGet(e) { const name = (e.parameter && e.parameter.sheet) || 'Reports'; const sh = SS.getSheetByName(name); if (!sh) return out([]); const rows = sh.getDataRange().getValues(); if (rows.length < 2) return out([]); const hdr = rows[0]; const data = rows.slice(1).map(r => { const o = {}; hdr.forEach((h, i) => o[h] = r[i]); return o; }); if (e.parameter && e.parameter.key) return out(data.find(r => String(r.Key) === String(e.parameter.key)) || null); return out(data); } function out(d) { return ContentService .createTextOutput(JSON.stringify(d)) .setMimeType(ContentService.MimeType.JSON); }
